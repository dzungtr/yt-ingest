# yt-ingest Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a CLI tool that ingests YouTube URLs, extracts structured learning notes via DeepSeek, and provides a locally queryable FAISS-backed knowledge base.

**Architecture:** Transcript fetching uses a chain-of-responsibility (youtube-transcript-api → yt-dlp auto-subs → Whisper). All DeepSeek calls go through a single `llm.py` wrapper that handles retries and logs prompt-cache hit/miss tokens. Extracted notes are rendered as Markdown files; a FAISS index over note chunks enables semantic search with DeepSeek-powered answers.

**Tech Stack:** Python 3.11+, typer, openai SDK (pointed at DeepSeek), pydantic v2, youtube-transcript-api, yt-dlp, webvtt-py, faster-whisper, sentence-transformers (BAAI/bge-large-en-v1.5), faiss-cpu, langchain-text-splitters, tiktoken, jinja2, python-dotenv, pytest, pytest-mock, ruff, mypy

```python
from __future__ import annotations
import hashlib
import json
import re
import subprocess
from pathlib import Path
from urllib.parse import parse_qs, urlparse

from yt_ingest.models import VideoMeta


def extract_video_id(url: str) -> str:
    """Parse a YouTube video ID from any standard YouTube URL format."""
    parsed = urlparse(url)
    video_id = ""
    if parsed.netloc in ("youtu.be",):
        video_id = parsed.path.lstrip("/").split("/")[0]
    elif "youtube.com" in parsed.netloc:
        if parsed.path.startswith("/shorts/"):
            video_id = parsed.path.split("/")[2]
        elif parsed.path.startswith("/embed/"):
            video_id = parsed.path.split("/")[2]
        else:
            qs = parse_qs(parsed.query)
            video_id = qs.get("v", [""])[0]
    if not re.match(r"^[A-Za-z0-9_-]{11}$", video_id):
        raise ValueError(f"Could not extract a valid video_id from URL: {url!r}")
    return video_id


def load_urls(path: Path) -> list[str]:
    """Load YouTube URLs from a file, skipping blank lines and comments, deduplicating."""
    seen: set[str] = set()
    result: list[str] = []
    for line in path.read_text().splitlines():
        line = line.strip()
        if not line or line.startswith("#"):
            continue
        if line not in seen:
            seen.add(line)
            result.append(line)
    return result


def seconds_to_mmss(seconds: float) -> str:
    """Convert a float number of seconds to mm:ss string (no zero-padding on minutes)."""
    total = int(seconds)
    m, s = divmod(total, 60)
    return f"{m}:{s:02d}"


def vtt_time_to_seconds(time_str: str) -> float:
    """Convert a VTT timestamp ('HH:MM:SS.mmm' or 'MM:SS.mmm') to seconds."""
    time_str = time_str.replace(",", ".")
    parts = time_str.split(":")
    if len(parts) == 3:
        h, m, s = parts
        return int(h) * 3600 + int(m) * 60 + float(s)
    m, s = parts
    return int(m) * 60 + float(s)


def file_checksum(path: Path) -> str:
    """SHA-256 hex digest of a file's contents."""
    return hashlib.sha256(path.read_bytes()).hexdigest()


def fetch_video_meta(url: str) -> VideoMeta:
    """Fetch video title, channel, and duration via yt-dlp --dump-json."""
    result = subprocess.run(
        ["yt-dlp", "--dump-json", url],
        capture_output=True,
        text=True,
        timeout=30,
    )
    if result.returncode != 0:
        raise RuntimeError(f"yt-dlp --dump-json failed: {result.stderr[:300]}")
    data: dict[str, object] = json.loads(result.stdout)
    return VideoMeta(
        title=str(data.get("title", "Unknown")),
        channel=str(data.get("uploader", data.get("channel", "Unknown"))),
        duration_seconds=int(data.get("duration", 0)),
    )
```

```python
from __future__ import annotations
import logging
from typing import Protocol, runtime_checkable

from yt_ingest.models import TranscriptCache

logger = logging.getLogger(__name__)


class FetchError(Exception):
    """Raised by any TranscriptFetcher when it cannot fetch a transcript."""


@runtime_checkable
class TranscriptFetcher(Protocol):
    name: str

    def fetch(self, video_id: str, url: str) -> TranscriptCache:
        ...


def run_fetcher_chain(
    video_id: str,
    url: str,
    fetchers: list[TranscriptFetcher],
) -> TranscriptCache:
    """Try each fetcher in order; return the first successful result.

    Logs ATTEMPT, SUCCESS, or FAIL for each fetcher so the caller can see
    exactly which path was taken.
    """
    if not fetchers:
        raise FetchError(f"No fetchers configured for {video_id}")

    last_error: Exception | None = None
    for fetcher in fetchers:
        logger.info("[%s] ATTEMPT %s", video_id, fetcher.name)
        try:
            result = fetcher.fetch(video_id, url)
            logger.info("[%s] SUCCESS via %s", video_id, fetcher.name)
            return result
        except FetchError as exc:
            logger.warning("[%s] FAIL %s: %s", video_id, fetcher.name, exc)
            last_error = exc

    raise FetchError(
        f"All fetchers failed for {video_id}"
    ) from last_error
```

```python
from __future__ import annotations
import pytest
from pathlib import Path
from unittest.mock import MagicMock
from yt_ingest.fetchers.base import FetchError, run_fetcher_chain
from yt_ingest.models import TranscriptSource


def _make_fetcher(name: str, raises: bool = False, cache=None):
    f = MagicMock()
    f.name = name
    if raises:
        f.fetch.side_effect = FetchError(f"{name} failed")
    else:
        f.fetch.return_value = cache
    return f


def test_chain_returns_first_success(sample_transcript_cache):
    f1 = _make_fetcher("first", raises=False, cache=sample_transcript_cache)
    f2 = _make_fetcher("second", raises=False, cache=sample_transcript_cache)
    result = run_fetcher_chain("dQw4w9WgXcQ", "https://youtu.be/dQw4w9WgXcQ", [f1, f2])
    assert result is sample_transcript_cache
    f1.fetch.assert_called_once()
    f2.fetch.assert_not_called()

    f1 = _make_fetcher("first", raises=True)

    result = run_fetcher_chain("dQw4w9WgXcQ", "https://youtu.be/dQw4w9WgXcQ", [f1, f2])
```

```python
import logging
import subprocess
import tempfile
from datetime import datetime, timezone

import webvtt

from yt_ingest.fetchers.base import FetchError
from yt_ingest.models import TranscriptCache, TranscriptSegment, TranscriptSource
from yt_ingest.utils import fetch_video_meta, vtt_time_to_seconds

logger = logging.getLogger(__name__)


class YtDlpFetcher:
    name = "ytdlp_subs"

    def fetch(self, video_id: str, url: str) -> TranscriptCache:
        meta = fetch_video_meta(url)

        with tempfile.TemporaryDirectory() as tmpdir:
            result = subprocess.run(
                [
                    "yt-dlp",
                    "--write-auto-sub",
                    "--skip-download",
                    "--sub-format", "vtt",
                    "--sub-langs", "en",
                    "--paths", tmpdir,
                    "-o", video_id,
                    url,
                ],
                capture_output=True,
                text=True,
                timeout=60,
            )
            vtt_files = list(Path(tmpdir).glob("*.vtt"))
            if not vtt_files:
                raise FetchError(
                    f"yt-dlp found no VTT subtitles for {video_id} "
                    f"(stderr: {result.stderr[:200]})"
                )
            segments = _parse_vtt(vtt_files[0])

        return TranscriptCache(
            video_id=video_id,
            url=url,
            source=TranscriptSource.YTDLP_SUBS,
            fetched_at=datetime.now(timezone.utc),
            title=meta.title,
            channel=meta.channel,
            duration_seconds=meta.duration_seconds,
            segments=segments,
        )


def _parse_vtt(path: Path) -> list[TranscriptSegment]:
    segments = []
    for caption in webvtt.read(str(path)):
        start = vtt_time_to_seconds(caption.start)
        end = vtt_time_to_seconds(caption.end)
        text = caption.text.strip()
        if text:
            segments.append(
                TranscriptSegment(text=text, start=start, duration=max(0.0, end - start))
            )
    return segments
```

```python
from __future__ import annotations
import logging
import time
from dataclasses import dataclass, field

from openai import APIError, OpenAI, RateLimitError

from yt_ingest.config import get_config

logger = logging.getLogger(__name__)

_client: OpenAI | None = None


def _get_client() -> OpenAI:
    global _client
    if _client is None:
        cfg = get_config()
        _client = OpenAI(
            api_key=cfg.deepseek_api_key,
            base_url="https://api.deepseek.com",
        )
    return _client


@dataclass
class _UsageAccumulator:
    input_tokens: int = 0
    output_tokens: int = 0
    cache_hit_tokens: int = 0
    cache_miss_tokens: int = 0


_usage = _UsageAccumulator()


def chat(
    model: str,
    messages: list[dict[str, str]],
    response_format: dict[str, str] | None = None,
    max_retries: int = 3,
) -> str:
    """Call DeepSeek chat completion with retry + token accounting."""
    client = _get_client()
    delay = 1.0
    for attempt in range(max_retries):
        try:
            kwargs: dict = {"model": model, "messages": messages}
            if response_format:
                kwargs["response_format"] = response_format
            resp = client.chat.completions.create(**kwargs)

            usage = resp.usage
            if usage:
                hit = getattr(usage, "prompt_cache_hit_tokens", 0) or 0
                miss = getattr(usage, "prompt_cache_miss_tokens", 0) or 0
                # Fallback: if DeepSeek fields absent, treat all input as miss
                if hit == 0 and miss == 0:
                    miss = usage.prompt_tokens
                _usage.input_tokens += usage.prompt_tokens
                _usage.output_tokens += usage.completion_tokens
                _usage.cache_hit_tokens += hit
                _usage.cache_miss_tokens += miss
                logger.info(
                    "[llm] model=%s in=%d out=%d cache_hit=%d cache_miss=%d",
                    model,
                    usage.prompt_tokens,
                    usage.completion_tokens,
                    hit,
                    miss,
                )

            return resp.choices[0].message.content or ""

        except (RateLimitError, APIError) as exc:
            if attempt == max_retries - 1:
                raise
            logger.warning(
                "[llm] attempt %d/%d failed (%s), retrying in %.0fs",
                attempt + 1,
                max_retries,
                exc,
                delay,
            )
            time.sleep(delay)
            delay *= 4

    raise RuntimeError("unreachable")  # pragma: no cover


def get_usage_summary() -> dict[str, object]:
    """Return accumulated token counts and a cost estimate (deepseek-chat pricing)."""
    # deepseek-chat pricing USD/1M tokens (verify at platform.deepseek.com/api-docs/pricing)
    INPUT_RATE = 0.27 / 1_000_000
    OUTPUT_RATE = 1.10 / 1_000_000
    CACHE_HIT_RATE = 0.07 / 1_000_000

    cost = (
        _usage.cache_miss_tokens * INPUT_RATE
        + _usage.cache_hit_tokens * CACHE_HIT_RATE
        + _usage.output_tokens * OUTPUT_RATE
    )
    return {
        "input_tokens": _usage.input_tokens,
        "output_tokens": _usage.output_tokens,
        "cache_hit_tokens": _usage.cache_hit_tokens,
        "cache_miss_tokens": _usage.cache_miss_tokens,
        "estimated_cost_usd": round(cost, 5),
    }
```

```python
from __future__ import annotations
import json
import logging
import tiktoken
from pathlib import Path

from jinja2 import Environment, FileSystemLoader

from yt_ingest.llm import chat
from yt_ingest.models import ExtractedContent, TranscriptCache
from yt_ingest.utils import seconds_to_mmss

logger = logging.getLogger(__name__)

TEMPLATE_DIR = Path(__file__).parent / "templates"
MAX_TRANSCRIPT_TOKENS = 80_000
EXTRACTION_MODEL = "deepseek-chat"

EXTRACTION_SYSTEM_PROMPT = """You are a learning assistant. Extract structured notes from the YouTube video transcript below.

Return a JSON object with exactly these fields:
- "summary": 3-5 sentence summary of the video (string)
- "key_claims": array of {text: string, timestamp_seconds: number} — 3-8 important claims with segment start times
- "frameworks_and_mental_models": array of strings — conceptual frameworks or mental models introduced
- "definitions": object mapping term to definition (string) — specialized vocabulary
- "worth_rewatching": array of {timestamp_seconds: number, reason: string} — moments worth revisiting
- "counterpoints_and_caveats": array of strings — limitations or caveats the speaker acknowledges
- "open_questions": array of strings — questions the video raises but doesn't fully answer

Rules:
- Use only timestamps that appear as segment start times in the transcript
- If a section has no relevant content, use an empty array or empty object
- Output raw JSON only — no markdown fences"""


def _build_transcript_text(transcript: TranscriptCache) -> str:
    lines = []
    for seg in transcript.segments:
        lines.append(f"[{seconds_to_mmss(seg.start)}] {seg.text}")
    return "\n".join(lines)


def _truncate_to_tokens(text: str, max_tokens: int) -> str:
    enc = tiktoken.get_encoding("cl100k_base")
    tokens = enc.encode(text)
    if len(tokens) <= max_tokens:
        return text
    logger.warning("Transcript exceeds %d tokens (%d), truncating", max_tokens, len(tokens))
    return enc.decode(tokens[:max_tokens])


def extract_note(transcript: TranscriptCache) -> str:
    """Extract structured notes and render to Markdown. Returns the full note text."""
    from datetime import datetime, timezone

    transcript_text = _build_transcript_text(transcript)
    transcript_text = _truncate_to_tokens(transcript_text, MAX_TRANSCRIPT_TOKENS)

    raw = chat(
        model=EXTRACTION_MODEL,
        messages=[
            {"role": "system", "content": EXTRACTION_SYSTEM_PROMPT},
            {"role": "user", "content": transcript_text},
        ],
        response_format={"type": "json_object"},
    )

    content = ExtractedContent.model_validate(json.loads(raw))

    env = Environment(loader=FileSystemLoader(str(TEMPLATE_DIR)), autoescape=False)
    env.filters["mmss"] = seconds_to_mmss
    template = env.get_template("note.md.j2")

    return template.render(
        video_id=transcript.video_id,
        url=transcript.url,
        title=transcript.title,
        channel=transcript.channel,
        duration_seconds=transcript.duration_seconds,
        transcript_source=transcript.source.value,
        extracted_at=datetime.now(timezone.utc).isoformat(),
        model=EXTRACTION_MODEL,
        content=content,
    )
```

```python
@app.command()
def extract() -> None:
    """Extract structured notes from cached transcripts."""
    import hashlib
    from yt_ingest.extract import extract_note
    from yt_ingest.models import TranscriptCache

    cfg = get_config()
    cfg.ensure_dirs()

    transcript_files = sorted(cfg.transcripts_dir.glob("*.json"))
    if not transcript_files:
        typer.echo("No cached transcripts found. Run 'yt-ingest fetch' first.")
        raise typer.Exit(1)

    ok = fail = skip = 0
    for tf in transcript_files:
        video_id = tf.stem
        note_path = cfg.notes_dir / f"{video_id}.md"
        checksum_path = cfg.notes_dir / f"{video_id}.md.checksum"
        current_hash = hashlib.sha256(tf.read_bytes()).hexdigest()

        if note_path.exists() and checksum_path.exists():
            if checksum_path.read_text().strip() == current_hash:
                logger.info("[%s] SKIP (note up-to-date)", video_id)
                skip += 1
                continue

        try:
            transcript = TranscriptCache.model_validate_json(tf.read_text())
            note_md = extract_note(transcript)
        except Exception as exc:
            typer.echo(f"FAIL {video_id}: {exc}", err=True)
            fail += 1
            continue

        note_path.write_text(note_md)
        checksum_path.write_text(current_hash)
        logger.info("[%s] EXTRACTED → %s", video_id, note_path)
        ok += 1

    typer.echo(f"\nDone: {ok} extracted, {skip} skipped, {fail} failed")
    if fail:
        raise typer.Exit(1)
```

```python
from __future__ import annotations
import json
import logging
from pathlib import Path

import faiss
import numpy as np
from langchain_text_splitters import RecursiveCharacterTextSplitter
from sentence_transformers import SentenceTransformer

logger = logging.getLogger(__name__)

EMBED_MODEL_NAME = "BAAI/bge-large-en-v1.5"
CHUNK_SIZE = 500
CHUNK_OVERLAP = 50

_embed_model: SentenceTransformer | None = None


def _get_embed_model() -> SentenceTransformer:
    global _embed_model
    if _embed_model is None:
        _embed_model = SentenceTransformer(EMBED_MODEL_NAME)
        logger.info("Loaded embedding model %s", EMBED_MODEL_NAME)
    return _embed_model


def _split_note(text: str) -> list[str]:
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=CHUNK_SIZE,
        chunk_overlap=CHUNK_OVERLAP,
        length_function=len,
    )
    return splitter.split_text(text)


def build_index(notes_dir: Path, index_dir: Path) -> None:
    """Embed all notes and write a FAISS index + chunk manifest to index_dir."""
    note_files = sorted(f for f in notes_dir.glob("*.md") if f.name != "_index.md")
    if not note_files:
        raise ValueError("No note files found. Run 'yt-ingest extract' first.")

    model = _get_embed_model()
    chunks: list[dict[str, str]] = []
    embeddings: list[np.ndarray] = []

    for note_path in note_files:
        video_id = note_path.stem
        for chunk_text in _split_note(note_path.read_text()):
            emb = model.encode(chunk_text, normalize_embeddings=True)
            chunks.append({"video_id": video_id, "text": chunk_text})
            embeddings.append(emb)

    logger.info("Indexing %d chunks from %d notes", len(chunks), len(note_files))
    dim = len(embeddings[0])
    index = faiss.IndexFlatIP(dim)  # inner product on unit vectors = cosine similarity
    index.add(np.array(embeddings, dtype=np.float32))

    index_dir.mkdir(parents=True, exist_ok=True)
    faiss.write_index(index, str(index_dir / "index.faiss"))
    (index_dir / "chunks.json").write_text(json.dumps(chunks, ensure_ascii=False))
    logger.info("Index written to %s", index_dir)


def query(question: str, index_dir: Path, top_k: int = 5) -> list[dict[str, object]]:
    """Retrieve the top-k most relevant chunks for a question."""
    model = _get_embed_model()
    q_emb = model.encode(question, normalize_embeddings=True)

    index = faiss.read_index(str(index_dir / "index.faiss"))
    chunks: list[dict[str, str]] = json.loads((index_dir / "chunks.json").read_text())

    scores, idxs = index.search(np.array([q_emb], dtype=np.float32), top_k)
    return [
        {"score": float(scores[0][i]), **chunks[int(idxs[0][i])]}
        for i in range(len(idxs[0]))
        if int(idxs[0][i]) >= 0
    ]
```

```python
def run(
    urls_file: str = typer.Argument(..., help="Path to file with YouTube URLs"),
    allow_whisper: bool = typer.Option(False, "--allow-whisper"),
) -> None:
    """Run the full pipeline: fetch → extract → index → synthesize."""
    import sys
    from yt_ingest.llm import get_usage_summary
    from yt_ingest.extract import extract_note
    from yt_ingest.retrieval import build_index
    from yt_ingest.synthesize import synthesize as _synthesize
    from yt_ingest.fetchers.base import FetchError, run_fetcher_chain
    from yt_ingest.fetchers.youtube_api import YouTubeTranscriptApiFetcher
    from yt_ingest.fetchers.ytdlp import YtDlpFetcher
    from yt_ingest.fetchers.whisper import WhisperFetcher
    from yt_ingest.models import TranscriptCache
    from yt_ingest.utils import extract_video_id, load_urls
    import hashlib

    cfg = get_config()
    cfg.ensure_dirs()
    urls = load_urls(Path(urls_file))

        if out_path.exists():

        try:
            transcript = run_fetcher_chain(video_id, url, fetchers)
            out_path.write_text(transcript.model_dump_json(indent=2))
        except FetchError as exc:
            typer.echo(f"FAIL fetch {video_id}: {exc}", err=True)

    # --- extract ---
    for tf in sorted(cfg.transcripts_dir.glob("*.json")):
        video_id = tf.stem
        note_path = cfg.notes_dir / f"{video_id}.md"
        checksum_path = cfg.notes_dir / f"{video_id}.md.checksum"
        current_hash = hashlib.sha256(tf.read_bytes()).hexdigest()
        if note_path.exists() and checksum_path.exists() and checksum_path.read_text().strip() == current_hash:
            continue
        try:
            transcript = TranscriptCache.model_validate_json(tf.read_text())
            note_md = extract_note(transcript)
            note_path.write_text(note_md)
            checksum_path.write_text(current_hash)
        except Exception as exc:
            typer.echo(f"FAIL extract {video_id}: {exc}", err=True)

    # --- index ---
    try:
        build_index(cfg.notes_dir, cfg.faiss_index_dir)
    except ValueError as exc:
        typer.echo(str(exc), err=True)

    # --- synthesize ---
    try:
        index_md = _synthesize(cfg.notes_dir)
        (cfg.notes_dir / "_index.md").write_text(index_md)
    except ValueError as exc:
        typer.echo(str(exc), err=True)

    # --- cost summary ---
    summary = get_usage_summary()
    print(
        f"\nToken usage: {summary['input_tokens']} in "
        f"({summary['cache_hit_tokens']} cache hit, {summary['cache_miss_tokens']} miss) "
        f"/ {summary['output_tokens']} out — "
        f"est. ${summary['estimated_cost_usd']:.4f}",
        file=sys.stderr,
    )
    typer.echo("Pipeline complete.")
```

## Transcript fallback chain

1. **youtube-transcript-api** — fastest, no auth, works when captions exist
2. **yt-dlp auto-subs** — YouTube's auto-generated captions, fetched as VTT
3. **Whisper** (`--allow-whisper` only) — downloads audio and transcribes locally with `faster-whisper large-v3`

Each attempt is logged. If all three fail for a URL, the error is printed and the pipeline continues with the next URL.

## Troubleshooting

### `TranscriptsDisabled` or `IpBlocked` errors

YouTube rate-limits transcript API requests by IP. This is not a bug in the code.

- The fetcher already retries with exponential backoff (1s → 4s → 16s)
- Run requests sequentially (the default) — parallel fetching triggers blocks faster
- If blocked, wait 15–30 minutes before retrying
- Use `yt-dlp` fallback or Whisper as alternatives

### GPU not found for Whisper

`faster-whisper` defaults to CUDA. If you don't have a GPU, edit `fetchers/whisper.py` and change `device="cuda"` to `device="cpu"` (expect ~4× slower transcription).
